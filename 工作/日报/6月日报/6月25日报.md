
### 3. 挂载数据盘（LVM）(测试服务器如果和数据库子是同一台服务器，就不用此操作)

```bash
yum install -y lvm2
pvcreate /dev/vdb /dev/vdc
vgcreate vgdata /dev/vdb
vgcreate vgfra /dev/vdc
lvcreate -l 100%VG -n lvdata vgdata
lvcreate -l 100%VG -n lvfra vgfra

mkfs.xfs -f /dev/vgdata/lvdata
mkfs.xfs -f /dev/vgfra/lvfra

mkdir -p /data /fra
mount /dev/vgdata/lvdata /data
mount /dev/vgfra/lvfra /fra

# 写入 fstab（自动挂载）
echo "$(blkid /dev/vgdata/lvdata | awk '{print $2}' | sed 's/\"//g') /data xfs defaults 0 0" >> /etc/fstab
echo "$(blkid /dev/vgfra/lvfra | awk '{print $2}' | sed 's/\"//g') /fra xfs defaults 0 0" >> /etc/fstab
```
### 4. 创建 4GB 虚拟内存（swap）

```bash
swapoff -a
rm -f /swapfile
dd if=/dev/zero of=/swapfile bs=1M count=4000
chmod 600 /swapfile
mkswap /swapfile
swapon /swapfile
echo '/swapfile none swap defaults 0 0' >> /etc/fstab
```

### 5.调整 /dev/shm 大小
```bash
echo 'tmpfs /dev/shm tmpfs defaults,size=16G 0 0' >> /etc/fstab
systemctl daemon-reload
mount -o remount /dev/shm
```
